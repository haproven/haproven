<script>
(() => {
    const input = document.getElementById("imageInput");
    const uploadBox = document.getElementById("uploadBox");
    const panel = document.getElementById("converterPanel");
    const preview = document.getElementById("imagePreview");
    const info = document.getElementById("imageInfo");

    const format = document.getElementById("outputFormat");
    const quality = document.getElementById("imageQuality");
    const qualityValue = document.getElementById("qualityValue");

    const convertBtn = document.getElementById("convertBtn");
    const downloadBtn = document.getElementById("downloadBtn");
    const chooseLocationBtn = document.getElementById("chooseLocationBtn");

    const fileNameInput = document.getElementById("fileName");
    const status = document.getElementById("statusMessage");
    const downloadStatus = document.getElementById("downloadStatus");

    let originalImage = null;
    let originalURL = null;
    let convertedBlob = null;
    let isConverting = false;

    const MAX_SIZE = 20 * 1024 * 1024;

    function getFileName() {
        let name = fileNameInput.value.trim();

        // Remove file extensions and invalid filename characters.
        name = name.replace(/\.[^.]*$/, "");
        name = name.replace(/[<>:"/\\|?*\x00-\x1F]/g, "-");
        name = name.replace(/[. ]+$/g, "").trim();

        return name || "haproven-image";
    }

    function getExtension() {
        return {
            "image/webp": "webp",
            "image/jpeg": "jpg",
            "image/png": "png"
        }[format.value];
    }

    function updateFilenameHint() {
        downloadStatus.textContent = convertedBlob
            ? `File name: ${getFileName()}.${getExtension()}`
            : "";
    }

    function clearConverted() {
        convertedBlob = null;
        downloadBtn.disabled = true;
        chooseLocationBtn.disabled = true;
        downloadStatus.textContent = "";
    }

    function loadImage(file) {
        if (!file) return;

        if (!["image/jpeg", "image/png", "image/webp"].includes(file.type)) {
            status.textContent = "Please select a JPG, PNG or WebP image.";
            return;
        }

        if (file.size > MAX_SIZE) {
            status.textContent = "Image is too large. Choose an image under 20 MB.";
            return;
        }

        clearConverted();

        const nextURL = URL.createObjectURL(file);
        const img = new Image();

        img.onload = () => {
            if (originalURL) URL.revokeObjectURL(originalURL);

            originalURL = nextURL;
            originalImage = img;
            preview.src = originalURL;

            info.textContent =
                `${file.name} · ${img.naturalWidth} × ${img.naturalHeight}px · ${(file.size / 1024).toFixed(1)} KB`;

            panel.classList.add("active");
            status.textContent = "Image ready to convert.";
            downloadStatus.textContent = "";
        };

        img.onerror = () => {
            URL.revokeObjectURL(nextURL);
            status.textContent = "Could not read this image.";
        };

        img.src = nextURL;
    }

    input.addEventListener("change", () => {
        loadImage(input.files[0]);
    });

    ["dragenter", "dragover"].forEach(eventName => {
        uploadBox.addEventListener(eventName, event => {
            event.preventDefault();
            uploadBox.classList.add("dragging");
        });
    });

    ["dragleave", "drop"].forEach(eventName => {
        uploadBox.addEventListener(eventName, event => {
            event.preventDefault();
            uploadBox.classList.remove("dragging");
        });
    });

    uploadBox.addEventListener("drop", event => {
        loadImage(event.dataTransfer.files[0]);
    });

    quality.addEventListener("input", () => {
        qualityValue.textContent = quality.value + "%";
    });

    format.addEventListener("change", clearConverted);
    quality.addEventListener("change", clearConverted);
    fileNameInput.addEventListener("input", updateFilenameHint);

    convertBtn.addEventListener("click", () => {
        if (!originalImage) {
            status.textContent = "Please upload an image first.";
            return;
        }

        if (isConverting) return;

        clearConverted();
        isConverting = true;
        convertBtn.disabled = true;
        status.textContent = "Converting image...";

        const canvas = document.createElement("canvas");
        canvas.width = originalImage.naturalWidth;
        canvas.height = originalImage.naturalHeight;

        const ctx = canvas.getContext("2d");

        if (!ctx) {
            isConverting = false;
            convertBtn.disabled = false;
            status.textContent = "Your browser could not create the image canvas.";
            return;
        }

        const mime = format.value;

        // JPEG cannot preserve transparent backgrounds.
        if (mime === "image/jpeg") {
            ctx.fillStyle = "#ffffff";
            ctx.fillRect(0, 0, canvas.width, canvas.height);
        }

        ctx.drawImage(originalImage, 0, 0);

        canvas.toBlob(blob => {
            isConverting = false;
            convertBtn.disabled = false;

            if (!blob) {
                status.textContent = "Conversion failed. Please try again.";
                return;
            }

            // Some browsers may not support a requested output format.
            if (blob.type !== mime) {
                status.textContent =
                    "This output format is not supported by your browser.";
                return;
            }

            convertedBlob = blob;
            downloadBtn.disabled = false;
            chooseLocationBtn.disabled = false;

            status.textContent =
                `Conversion complete! Image size: ${(blob.size / 1024).toFixed(1)} KB.`;

            updateFilenameHint();
        }, mime, Number(quality.value) / 100);
    });

    // Option 1: Download directly using the browser's normal download flow.
    downloadBtn.addEventListener("click", () => {
        if (!convertedBlob) {
            downloadStatus.textContent = "Please convert the image first.";
            return;
        }

        const url = URL.createObjectURL(convertedBlob);
        const link = document.createElement("a");

        link.href = url;
        link.download = `${getFileName()}.${getExtension()}`;

        document.body.appendChild(link);
        link.click();
        link.remove();

        setTimeout(() => URL.revokeObjectURL(url), 2000);

        downloadStatus.textContent =
            "Download started. Check your browser's Downloads folder.";
    });

    // Option 2: Let the user choose a save location, where supported.
    chooseLocationBtn.addEventListener("click", async () => {
        if (!convertedBlob) {
            downloadStatus.textContent = "Please convert the image first.";
            return;
        }

        const name = `${getFileName()}.${getExtension()}`;

        try {
            if ("showSaveFilePicker" in window) {
                const handle = await window.showSaveFilePicker({
                    suggestedName: name,
                    types: [{
                        description: "Converted Image",
                        accept: {
                            [format.value]: ["." + getExtension()]
                        }
                    }]
                });

                const writable = await handle.createWritable();

                await writable.write(convertedBlob);
                await writable.close();

                downloadStatus.textContent = "Image saved to your chosen location.";
            } else {
                // Fallback for browsers without a save-location picker.
                const url = URL.createObjectURL(convertedBlob);
                const link = document.createElement("a");

                link.href = url;
                link.download = name;
                document.body.appendChild(link);
                link.click();
                link.remove();

                setTimeout(() => URL.revokeObjectURL(url), 2000);

                downloadStatus.textContent =
                    "Your browser does not support choosing a save location. Normal download started.";
            }
        } catch (error) {
            if (error.name === "AbortError") {
                downloadStatus.textContent = "Save cancelled.";
            } else {
                downloadStatus.textContent =
                    "Could not save the image. Try the normal Download Image option.";
            }
        }
    });

    document.getElementById("removeImage").addEventListener("click", () => {
        input.value = "";
        originalImage = null;

        if (originalURL) URL.revokeObjectURL(originalURL);
        originalURL = null;

        clearConverted();

        preview.removeAttribute("src");
        info.textContent = "";
        status.textContent = "";
        panel.classList.remove("active");
    });
})();
</script>
